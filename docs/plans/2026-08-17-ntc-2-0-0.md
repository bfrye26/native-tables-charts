	$out = '';
	foreach ( array_slice( (array) ( $c['referenceLines'] ?? array() ), 0, 5 ) as $rl ) {
		$v = (float) ( $rl['value'] ?? 0 );
		if ( $v < 0 || $max <= 0 ) { continue; }

	return $out;
}
